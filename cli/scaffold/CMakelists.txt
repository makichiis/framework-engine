cmake_minimum_required(VERSION 3.1...3.29)

project(framework-demo LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)

add_compile_options(-Werror -O3 -Wall)

include_directories(${CMAKE_SOURCE_DIR}/lib/framework-core/include)
add_subdirectory(lib/framework-core)

add_executable(framework-demo main.cpp)
target_link_libraries(framework-demo PRIVATE glfw glad framework)
